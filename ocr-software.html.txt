<script>
(function(){
'use strict';
const fileInput = document.getElementById('fileInput');
const uploadZone = document.getElementById('uploadZone');
const resultBox = document.getElementById('resultBox');
const copyBtn = document.getElementById('copyBtn');
const clearBtn = document.getElementById('clearBtn');
const statusMsg = document.getElementById('statusMsg');
const themeToggle = document.getElementById('themeToggle');

function setStatus(msg, isError=false) {
statusMsg.textContent = msg;
statusMsg.style.borderLeftColor = isError ? '#ff3333' : '#ffd200';
}

uploadZone.addEventListener('click', () => fileInput.click());
fileInput.addEventListener('change', (e) => {
if (e.target.files[0]) processImage(e.target.files[0]);
});
uploadZone.addEventListener('dragover', (e) => { e.preventDefault(); uploadZone.style.borderColor = 'var(--accent)'; });
uploadZone.addEventListener('dragleave', () => { uploadZone.style.borderColor = 'var(--border)'; });
uploadZone.addEventListener('drop', (e) => {
e.preventDefault();
uploadZone.style.borderColor = 'var(--border)';
if (e.dataTransfer.files[0]) processImage(e.dataTransfer.files[0]);
});

async function processImage(file) {
setStatus('⏳ Extracting text...');
try {
const reader = new FileReader();
reader.onload = async (e) => {
const img = new Image();
img.onload = async () => {
const canvas = document.createElement('canvas');
canvas.width = img.width;
canvas.height = img.height;
const ctx = canvas.getContext('2d');
ctx.drawImage(img, 0, 0);
const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);

// Use Tesseract.js for OCR
const { data: { text } } = await Tesseract.recognize(
canvas,
'eng+hin', // English + Hindi
{ logger: m => console.log(m) }
);
resultBox.textContent = text.trim() || 'No text found.';
copyBtn.disabled = false;
setStatus('✅ Text extracted successfully!');
};
img.src = e.target.result;
};
reader.readAsDataURL(file);
} catch(err) {
setStatus('❌ Error: ' + err.message, true);
}
}

copyBtn.addEventListener('click', () => {
navigator.clipboard.writeText(resultBox.textContent);
setStatus('📋 Text copied to clipboard!');
});

clearBtn.addEventListener('click', () => {
resultBox.textContent = '📝 Extracted text will appear here...';
copyBtn.disabled = true;
setStatus('🗑 Cleared');
});

themeToggle.addEventListener('click', () => {
document.body.classList.toggle('light-theme');
themeToggle.textContent = document.body.classList.contains('light-theme') ? '☀️ Light' : '🌙 Dark';
});

setStatus('✅ Upload an image to extract text.');
})();
</script>
