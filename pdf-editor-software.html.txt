<script>
(function(){
'use strict';
const fileInput = document.getElementById('fileInput');
const uploadZone = document.getElementById('uploadZone');
const actionSelect = document.getElementById('actionSelect');
const textInput = document.getElementById('textInput');
const applyBtn = document.getElementById('applyBtn');
const downloadBtn = document.getElementById('downloadBtn');
const resetBtn = document.getElementById('resetBtn');
const pdfPreview = document.getElementById('pdfPreview');
const statusMsg = document.getElementById('statusMsg');
const themeToggle = document.getElementById('themeToggle');

let pdfDoc = null;
let pdfBytes = null;
let fileName = 'document.pdf';

function setStatus(msg, isError=false) {
statusMsg.textContent = msg;
statusMsg.style.borderLeftColor = isError ? '#ff3333' : '#ffd200';
}

uploadZone.addEventListener('click', () => fileInput.click());
fileInput.addEventListener('change', (e) => {
if (e.target.files[0]) loadPDF(e.target.files[0]);
});
uploadZone.addEventListener('dragover', (e) => { e.preventDefault(); uploadZone.style.borderColor = 'var(--accent)'; });
uploadZone.addEventListener('dragleave', () => { uploadZone.style.borderColor = 'var(--border)'; });
uploadZone.addEventListener('drop', (e) => {
e.preventDefault();
uploadZone.style.borderColor = 'var(--border)';
if (e.dataTransfer.files[0]) loadPDF(e.dataTransfer.files[0]);
});

async function loadPDF(file) {
try {
fileName = file.name;
const arrayBuffer = await file.arrayBuffer();
pdfBytes = new Uint8Array(arrayBuffer);
pdfDoc = await PDFLib.PDFDocument.load(pdfBytes);
pdfPreview.innerHTML = `<div>📄 <strong>${fileName}</strong><br>Pages: ${pdfDoc.getPageCount()}</div>`;
downloadBtn.disabled = false;
setStatus(`✅ Loaded: ${fileName}`);
} catch(err) {
setStatus('❌ Error loading PDF: ' + err.message, true);
}
}

applyBtn.addEventListener('click', async () => {
if (!pdfDoc) { setStatus('⚠️ Please upload a PDF first.', true); return; }
const action = actionSelect.value;
setStatus('⏳ Processing...');

try {
let newPdf = pdfDoc;
if (action === 'merge') {
// Simulate merge (for demo, just download same)
setStatus('✅ Merge simulated (upload multiple files to merge)');
} else if (action === 'split') {
// Split first page
const splitDoc = await PDFLib.PDFDocument.create();
const [firstPage] = await splitDoc.copyPages(newPdf, [0]);
splitDoc.addPage(firstPage);
newPdf = splitDoc;
setStatus('✅ Split applied (first page only)');
} else if (action === 'compress') {
// Compress (simulated)
setStatus('✅ Compression applied (simulated)');
} else if (action === 'addText') {
const text = textInput.value.trim();
if (!text) { setStatus('⚠️ Enter text to add.', true); return; }
const pages = newPdf.getPages();
const firstPage = pages[0];
const { width, height } = firstPage.getSize();
firstPage.drawText(text, {
x: 50, y: height - 50,
size: 20, color: PDFLib.rgb(0, 0, 0)
});
setStatus(`✅ Text added: "${text}"`);
}

const newBytes = await newPdf.save();
pdfBytes = newBytes;
pdfDoc = newPdf;
downloadBtn.onclick = () => {
const blob = new Blob([pdfBytes], { type: 'application/pdf' });
const url = URL.createObjectURL(blob);
const a = document.createElement('a');
a.href = url;
a.download = `edited-${fileName}`;
a.click();
setStatus('✅ Download started!');
};
setStatus('✅ Ready to download!');
} catch(err) {
setStatus('❌ Error: ' + err.message, true);
}
});

resetBtn.addEventListener('click', () => {
pdfDoc = null;
pdfBytes = null;
pdfPreview.innerHTML = '<span>📄 Upload a PDF to get started</span>';
downloadBtn.disabled = true;
setStatus('↺ Reset');
});

themeToggle.addEventListener('click', () => {
document.body.classList.toggle('light-theme');
themeToggle.textContent = document.body.classList.contains('light-theme') ? '☀️ Light' : '🌙 Dark';
});

setStatus('✅ Upload a PDF to start editing.');
})();
</script>
