<script>
(function(){
'use strict';
const board = document.getElementById('board');
const statusMsg = document.getElementById('statusMsg');
const themeToggle = document.getElementById('themeToggle');
const addColumnBtn = document.getElementById('addColumnBtn');
const clearBoardBtn = document.getElementById('clearBoardBtn');

let columns = JSON.parse(localStorage.getItem('kanbanColumns')) || [
{ id: 'col1', title: 'To Do', tasks: [] },
{ id: 'col2', title: 'In Progress', tasks: [] },
{ id: 'col3', title: 'Done', tasks: [] }
];
let taskIdCounter = Date.now();

function saveState() { localStorage.setItem('kanbanColumns', JSON.stringify(columns)); }

function renderBoard() {
board.innerHTML = '';
columns.forEach((col, colIdx) => {
const colDiv = document.createElement('div');
colDiv.className = 'column';
colDiv.dataset.colId = col.id;
colDiv.innerHTML = `
<h3>${col.title} <span style="font-size:12px;color:var(--text2)">(${col.tasks.length})</span></h3>
<div class="column-tasks" data-col="${col.id}">
${col.tasks.map((task, tIdx) => `
<div class="task" draggable="true" data-col="${col.id}" data-task="${task.id}">
<div class="title">${task.title}</div>
${task.desc ? `<div class="desc">${task.desc}</div>` : ''}
<div class="actions">
<button class="edit-task" data-col="${col.id}" data-task="${task.id}">✏️</button>
<button class="delete-task" data-col="${col.id}" data-task="${task.id}">✕</button>
</div>
</div>
`).join('')}
</div>
<div class="add-task">
<input type="text" placeholder="Task title" class="task-title" data-col="${col.id}" />
<textarea placeholder="Description (optional)" class="task-desc" data-col="${col.id}"></textarea>
<button class="btn btn-primary add-task-btn" data-col="${col.id}">+ Add Task</button>
</div>
`;
board.appendChild(colDiv);
});

// Drag & Drop
document.querySelectorAll('.task').forEach(task => {
task.addEventListener('dragstart', handleDragStart);
task.addEventListener('dragend', handleDragEnd);
});
document.querySelectorAll('.column-tasks').forEach(col => {
col.addEventListener('dragover', handleDragOver);
col.addEventListener('drop', handleDrop);
});

// Add task buttons
document.querySelectorAll('.add-task-btn').forEach(btn => {
btn.addEventListener('click', (e) => {
const colId = btn.dataset.col;
const titleInput = document.querySelector(`.task-title[data-col="${colId}"]`);
const descInput = document.querySelector(`.task-desc[data-col="${colId}"]`);
const title = titleInput.value.trim();
if (!title) { setStatus('⚠️ Please enter a task title', true); return; }
const col = columns.find(c => c.id === colId);
if (col) {
col.tasks.push({ id: `task_${taskIdCounter++}`, title, desc: descInput.value.trim() });
saveState();
renderBoard();
setStatus(`✅ Task added: "${title}"`);
titleInput.value = '';
descInput.value = '';
}
});
});

// Delete task
document.querySelectorAll('.delete-task').forEach(btn => {
btn.addEventListener('click', (e) => {
const colId = btn.dataset.col;
const taskId = btn.dataset.task;
const col = columns.find(c => c.id === colId);
if (col) {
col.tasks = col.tasks.filter(t => t.id !== taskId);
saveState();
renderBoard();
setStatus('🗑 Task deleted');
}
});
});

// Edit task (simple prompt for demo)
document.querySelectorAll('.edit-task').forEach(btn => {
btn.addEventListener('click', (e) => {
const colId = btn.dataset.col;
const taskId = btn.dataset.task;
const col = columns.find(c => c.id === colId);
if (col) {
const task = col.tasks.find(t => t.id === taskId);
if (task) {
const newTitle = prompt('Edit title:', task.title);
if (newTitle !== null && newTitle.trim() !== '') {
task.title = newTitle.trim();
const newDesc = prompt('Edit description:', task.desc);
if (newDesc !== null) task.desc = newDesc.trim();
saveState();
renderBoard();
setStatus('✏️ Task updated');
}
}
}
});
});
}

// Drag & Drop handlers
let draggedTask = null;
function handleDragStart(e) {
draggedTask = e.target;
e.dataTransfer.setData('text/plain', JSON.stringify({
colId: e.target.dataset.col,
taskId: e.target.dataset.task
}));
}
function handleDragEnd() { draggedTask = null; }
function handleDragOver(e) { e.preventDefault(); }
function handleDrop(e) {
e.preventDefault();
const targetCol = e.currentTarget.dataset.col;
const data = JSON.parse(e.dataTransfer.getData('text/plain'));
const sourceColId = data.colId;
const taskId = data.taskId;
if (sourceColId === targetCol) return;
const sourceCol = columns.find(c => c.id === sourceColId);
const targetColObj = columns.find(c => c.id === targetCol);
if (!sourceCol || !targetColObj) return;
const taskIndex = sourceCol.tasks.findIndex(t => t.id === taskId);
if (taskIndex === -1) return;
const task = sourceCol.tasks.splice(taskIndex, 1)[0];
targetColObj.tasks.push(task);
saveState();
renderBoard();
setStatus(`✅ Moved task to "${targetColObj.title}"`);
}

function setStatus(msg, isError=false) {
statusMsg.textContent = msg;
statusMsg.style.borderLeftColor = isError ? '#ff3333' : '#ffd200';
}

// Add column
addColumnBtn.addEventListener('click', () => {
const title = prompt('Enter column name:');
if (title && title.trim()) {
columns.push({ id: `col_${Date.now()}`, title: title.trim(), tasks: [] });
saveState();
renderBoard();
setStatus(`✅ Column added: "${title.trim()}"`);
}
});

// Clear all
clearBoardBtn.addEventListener('click', () => {
if (confirm('Delete all columns and tasks?')) {
columns = [];
saveState();
renderBoard();
setStatus('🗑 Board cleared');
}
});

// Theme
themeToggle.addEventListener('click', () => {
document.body.classList.toggle('light-theme');
themeToggle.textContent = document.body.classList.contains('light-theme') ? '☀️ Light' : '🌙 Dark';
});

// Init
renderBoard();
setStatus('✅ Ready! Add tasks and drag to move.');
})();
</script>
