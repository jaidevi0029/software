// ---- Helper: save history ----
function pushHistory() {
if (!canvasWidth || !canvasHeight) return;
const data = ctx.getImageData(0, 0, canvasWidth, canvasHeight);
history = history.slice(0, historyIndex + 1);
history.push(data);
if (history.length > MAX_HISTORY) history.shift();
historyIndex = history.length - 1;
updateHistoryUI();
updateUndoRedo();
}

// ---- Helper: restore history ----
function restoreHistory(index) {
if (index < 0 || index >= history.length) return;
historyIndex = index;
ctx.putImageData(history[index], 0, 0);
updateHistoryUI();
updateUndoRedo();
}

// ---- Update UI ----
function updateUndoRedo() {

redoBtn.disabled = historyIndex >= history.length - 1;
}

tempCanvas.height = 28;

});
}

function applyAdjustments() {
if (!originalImage) return;
const brightness = parseFloat(document.getElementById('slider_brightness')?.value || 0);
const contrast = parseFloat(document.getElementById('slider_contrast')?.value || 0);
const saturation = parseFloat(document.getElementById('slider_saturation')?.value || 0);
const hue = parseFloat(document.getElementById('slider_hue')?.value || 0);
const blur = parseFloat(document.getElementById('slider_blur')?.value || 0);
const sharpen = parseFloat(document.getElementById('slider_sharpen')?.value || 0);

ctx.clearRect(0, 0, canvasWidth, canvasHeight);
ctx.drawImage(originalImage, 0, 0);

const imageData = ctx.getImageData(0, 0, canvasWidth, canvasHeight);
const data = imageData.data;

for (let i = 0; i < data.length; i += 4) {
let r = data[i];
let g = data[i+1];
let b = data[i+2];

r = Math.min(255, Math.max(0, r + brightness * 2.55));
g = Math.min(255, Math.max(0, g + brightness * 2.55));
b = Math.min(255, Math.max(0, b + brightness * 2.55));

const factor = (259 * (contrast + 255)) / (255 * (259 - contrast));
r = Math.min(255, Math.max(0, factor * (r - 128) + 128));
g = Math.min(255, Math.max(0, factor * (g - 128) + 128));
b = Math.min(255, Math.max(0, factor * (b - 128) + 128));

const gray = 0.299 * r + 0.587 * g + 0.114 * b;
const satFactor = 1 + saturation / 100;
r = Math.min(255, Math.max(0, gray + (r - gray) * satFactor));
g = Math.min(255, Math.max(0, gray + (g - gray) * satFactor));
b = Math.min(255, Math.max(0, gray + (b - gray) * satFactor));

if (hue !== 0) {
const h = (hue / 180) * Math.PI;
const cos = Math.cos(h);
const sin = Math.sin(h);
const newR = r * cos - g * sin;
const newG = r * sin + g * cos;
r = Math.min(255, Math.max(0, newR));
g = Math.min(255, Math.max(0, newG));
}

data[i] = r;
data[i+1] = g;
data[i+2] = b;
}
ctx.putImageData(imageData, 0, 0);

if (blur > 0) {
ctx.filter = `blur(${blur}px)`;
ctx.drawImage(canvas, 0, 0);
ctx.filter = 'none';
}

if (sharpen > 0) {
setStatus('⚡ Sharpen applied (simulated)');
}

clearTimeout(window._adjustTimer);
window._adjustTimer = setTimeout(() => {
pushHistory();
}, 300);
}

function rotateCanvas(deg) {
const tempCanvas = document.createElement('canvas');
const tempCtx = tempCanvas.getContext('2d');
const w = canvasWidth, h = canvasHeight;
if (deg === 90 || deg === -90) {
tempCanvas.width = h;
tempCanvas.height = w;
tempCtx.translate(h/2, w/2);
tempCtx.rotate(deg * Math.PI / 180);
tempCtx.drawImage(canvas, -w/2, -h/2);
canvasWidth = h;
canvasHeight = w;
} else {
tempCanvas.width = w;
tempCanvas.height = h;
tempCtx.drawImage(canvas, 0, 0);
}
canvas.width = canvasWidth;
canvas.height = canvasHeight;
ctx.drawImage(tempCanvas, 0, 0);
pushHistory();
setStatus(`✅ Rotated ${deg}°`);
}

function flipCanvas(dir) {
const tempCanvas = document.createElement('canvas');
tempCanvas.width = canvasWidth;
tempCanvas.height = canvasHeight;
const tempCtx = tempCanvas.getContext('2d');
if (dir === 'h') {
tempCtx.translate(canvasWidth, 0);
tempCtx.scale(-1, 1);
} else {
tempCtx.translate(0, canvasHeight);
tempCtx.scale(1, -1);
}
tempCtx.drawImage(canvas, 0, 0);
ctx.clearRect(0, 0, canvasWidth, canvasHeight);
ctx.drawImage(tempCanvas, 0, 0);
pushHistory();
setStatus(`✅ Flipped ${dir === 'h' ? 'Horizontal' : 'Vertical'}`);
}

// ---- FILTER CONTROLS ----
function buildFilterControls() {
const filters = ['Sepia', 'Grayscale', 'Vintage', 'Invert', 'Neon', 'Noir'];
const div = document.createElement('div');
div.className = 'control-group';
div.style.display = 'flex';
div.style.flexWrap = 'wrap';
div.style.gap = '6px';
filters.forEach(f => {

btn.style.fontSize = '12px';

div.appendChild(btn);
});
controlsContainer.appendChild(div);
}

function applyFilter(filter) {
if (!originalImage) return;
const imageData = ctx.getImageData(0, 0, canvasWidth, canvasHeight);
const data = imageData.data;
for (let i = 0; i < data.length; i += 4) {
let r = data[i], g = data[i+1], b = data[i+2];
switch(filter) {
case 'Sepia':
data[i] = Math.min(255, r * 0.393 + g * 0.769 + b * 0.189);
data[i+1] = Math.min(255, r * 0.349 + g * 0.686 + b * 0.168);
data[i+2] = Math.min(255, r * 0.272 + g * 0.534 + b * 0.131);
break;
case 'Grayscale':
const gray = 0.299*r + 0.587*g + 0.114*b;
data[i] = data[i+1] = data[i+2] = gray;
break;
case 'Invert':
data[i] = 255 - r;
data[i+1] = 255 - g;
data[i+2] = 255 - b;
break;
case 'Vintage':
data[i] = Math.min(255, r * 0.9 + 20);
data[i+1] = Math.min(255, g * 0.7 + 10);
data[i+2] = Math.min(255, b * 0.5 + 30);
break;
case 'Neon':
data[i] = Math.min(255, r * 1.5);
data[i+1] = Math.min(255, g * 0.8);
data[i+2] = Math.min(255, b * 1.8);
break;
case 'Noir':
const gray2 = 0.299*r + 0.587*g + 0.114*b;
data[i] = data[i+1] = data[i+2] = Math.min(255, gray2 * 1.2);
break;
}
}
ctx.putImageData(imageData, 0, 0);
pushHistory();
setStatus(`✅ Applied ${filter} filter`);
}

// ---- Canvas drawing events ----
let isMouseDown = false;
let lastX = 0, lastY = 0;

canvas.addEventListener('mousedown', (e) => {
if (currentTool !== 'draw') return;
const rect = canvas.getBoundingClientRect();
const scaleX = canvas.width / rect.width;
const scaleY = canvas.height / rect.height;
lastX = (e.clientX - rect.left) * scaleX;
lastY = (e.clientY - rect.top) * scaleY;
isMouseDown = true;
isDrawing = true;
ctx.beginPath();
ctx.arc(lastX, lastY, drawSize/2, 0, Math.PI*2);
ctx.fillStyle = drawColor;
ctx.fill();
});

canvas.addEventListener('mousemove', (e) => {
if (!isMouseDown || currentTool !== 'draw') return;
const rect = canvas.getBoundingClientRect();
const scaleX = canvas.width / rect.width;
const scaleY = canvas.height / rect.height;
const x = (e.clientX - rect.left) * scaleX;
const y = (e.clientY - rect.top) * scaleY;
ctx.beginPath();
ctx.moveTo(lastX, lastY);
ctx.lineTo(x, y);
ctx.strokeStyle = drawColor;
ctx.lineWidth = drawSize;
ctx.lineCap = 'round';
ctx.stroke();
lastX = x;
lastY = y;
});

canvas.addEventListener('mouseup', () => {
if (isMouseDown && currentTool === 'draw') {
isMouseDown = false;
isDrawing = false;
pushHistory();
}
});
canvas.addEventListener('mouseleave', () => {
if (isMouseDown) {
isMouseDown = false;
isDrawing = false;
pushHistory();
}
});
